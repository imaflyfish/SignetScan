cmake_minimum_required(VERSION 3.24)
project(SignetScan VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SIGNET_SCAN_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(SIGNET_SCAN_FUZZ "Build the libFuzzer parser driver" OFF)
find_package(nlohmann_json 3.12 REQUIRED)
find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)
find_package(PkgConfig REQUIRED)
pkg_check_modules(PLIST REQUIRED IMPORTED_TARGET libplist-2.0>=2.7)
pkg_check_modules(PCRE2 REQUIRED IMPORTED_TARGET libpcre2-8>=10.40)
add_library(signet_scan
  src/input/byte_loader.cpp src/binary/mach_image_set.cpp
  src/signing/signature_blob.cpp src/entitlements/grant_document.cpp
  src/policy/audit_rules.cpp src/bundle/seal_manifest.cpp
  src/app/audit_session.cpp src/report/writers.cpp)
target_include_directories(signet_scan PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_features(signet_scan PUBLIC cxx_std_20)
target_link_libraries(signet_scan PUBLIC nlohmann_json::nlohmann_json PRIVATE OpenSSL::Crypto PkgConfig::PLIST PkgConfig::PCRE2)
target_compile_options(signet_scan PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(SIGNET_SCAN_SANITIZE)
  target_compile_options(signet_scan PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(signet_scan PUBLIC -fsanitize=address,undefined)
endif()
add_executable(signet-scan app/cli_main.cpp)
target_link_libraries(signet-scan PRIVATE signet_scan)
add_executable(scan-verify-system tools/system_probe.cpp)
target_link_libraries(scan-verify-system PRIVATE signet_scan)
include(CTest)
if(BUILD_TESTING)
  add_executable(scan-core-tests tests/unit/core_contract.cpp)
  target_link_libraries(scan-core-tests PRIVATE signet_scan PkgConfig::PLIST)
  add_test(NAME core-contract COMMAND scan-core-tests)
  add_executable(scan-seal-tests tests/unit/seal_contract.cpp)
  target_link_libraries(scan-seal-tests PRIVATE signet_scan PkgConfig::PLIST)
  add_test(NAME resource-contract COMMAND scan-seal-tests)
  add_executable(scan-rule-tests tests/unit/rule_contract.cpp)
  target_link_libraries(scan-rule-tests PRIVATE signet_scan)
  add_test(NAME rule-contract COMMAND scan-rule-tests)
  add_executable(scan-boundary-tests tests/unit/boundary_contract.cpp)
  target_link_libraries(scan-boundary-tests PRIVATE signet_scan)
  add_test(NAME boundary-contract COMMAND scan-boundary-tests)
  add_executable(scan-integration tests/integration/macos_live.cpp)
  target_link_libraries(scan-integration PRIVATE signet_scan)
  add_test(NAME macos-integration COMMAND scan-integration $<TARGET_FILE:signet-scan>)
  set_tests_properties(macos-integration PROPERTIES SKIP_RETURN_CODE 77)
endif()
if(SIGNET_SCAN_FUZZ)
  target_compile_options(signet_scan PRIVATE -fsanitize=fuzzer-no-link)
  add_executable(scan-fuzz tests/fuzz/decoder_driver.cpp)
  target_compile_options(scan-fuzz PRIVATE -fsanitize=fuzzer)
  target_link_options(scan-fuzz PRIVATE -fsanitize=fuzzer)
  target_link_libraries(scan-fuzz PRIVATE signet_scan)
endif()
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/SignetScanConfig.cmake.in
  ${CMAKE_CURRENT_BINARY_DIR}/SignetScanConfig.cmake
  INSTALL_DESTINATION lib/cmake/SignetScan)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/SignetScanConfigVersion.cmake
  COMPATIBILITY SameMajorVersion)
install(TARGETS signet-scan signet_scan EXPORT SignetScanTargets RUNTIME DESTINATION bin ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(EXPORT SignetScanTargets NAMESPACE SignetScan:: DESTINATION lib/cmake/SignetScan)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/SignetScanConfig.cmake
  ${CMAKE_CURRENT_BINARY_DIR}/SignetScanConfigVersion.cmake DESTINATION lib/cmake/SignetScan)
install(FILES README.md LICENSE THIRD_PARTY_NOTICES.md dependencies.lock.json
  DESTINATION share/doc/SignetScan)
install(DIRECTORY docs licenses validation examples DESTINATION share/doc/SignetScan)
